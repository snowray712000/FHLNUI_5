## Step 1：實作（最小可行版本）

新增檔：`preprocessCaseToLines.es2023.js`

```js
/**
 * 將 API 回傳的一個 case 文字，轉成「可解析」的 lines[]
 * - 合併排版造成的續行（縮排更深、且不是新條目開頭）
 * - 表格線（┌│─┘ 等）不合併，保留原行
 * @param {string} raw
 * @returns {string[]}
 */
export function preprocessCaseToLines(raw) {
  if (typeof raw !== 'string') return [];

  const lines = raw.replace(/\r\n/g, '\n').split('\n');

  /** @type {{text:string, indent:number}[]} */
  const out = [];
  let prev = null;

  for (const line of lines) {
    const trimmedRight = line.replace(/\s+$/g, '');
    const content = trimmedRight.trim();

    // 跳過空行（你若想保留空行語意，之後再加模式）
    if (!content) continue;

    const indent = (line.match(/^\s*/) || [''])[0].length;

    // 表格線偵測：遇到這些符號，一律視為「不可合併的 raw line」
    const isTableLine = /[┌┬┐├┼┤└┴┘│─]/.test(content) || /-{5,}/.test(content);

    // 新條目開頭偵測（先用「保守」版本）
    // 你之後可再擴充：零、壹、（一）、1. A. (1) a. ◎ ● ○ ☆ 【註1】等
    const isNewItem =
      /^[◎●○☆]/.test(content) ||
      /^[A-Za-z]\./.test(content) ||
      /^\(\d+\)/.test(content) ||
      /^\d+\./.test(content) ||
      /^（[一二三四五六七八九十]+）/.test(content) ||
      /^[零壹貳參肆伍陸柒捌玖拾]+、/.test(content) ||
      /^【註\d+】/.test(content);

    // 合併條件：有 prev、縮排更深、不是表格行、且這行不是新條目
    if (prev && indent > prev.indent && !isTableLine && !isNewItem) {
      prev.text += content; // 這裡先直接黏起來（不加空白），符合你「自\n發」需求
      continue;
    }

    prev = { text: content, indent };
    out.push(prev);
  }

  return out.map(x => x.text);
}
```
